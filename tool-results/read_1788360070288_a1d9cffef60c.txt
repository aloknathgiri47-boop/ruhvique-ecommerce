   711→function SecondaryStat({
   712→  label,
   713→  value,
   714→  growth,
   715→  icon: Icon,
   716→  color,
   717→  href,
   718→}: {
   719→  label: string;
   720→  value: string;
   721→  growth?: number;
   722→  icon: any;
   723→  color: string;
   724→  href?: string;
   725→}) {
   726→  const content = (
   727→    <Card className="p-4 hover:shadow-sm transition-shadow">
   728→      <div className="flex items-center gap-3">
   729→        <div className={`inline-flex h-9 w-9 items-center justify-center rounded-md ${color}`}>
   730→          <Icon className="h-4 w-4" />
   731→        </div>
   732→        <div className="flex-1 min-w-0">
   733→          <p className="text-xs uppercase tracking-wider text-muted-foreground">{label}</p>
   734→          <p className="text-lg font-black">{value}</p>
   735→        </div>
   736→        {growth !== undefined && (
   737→          <div className={`inline-flex items-center gap-0.5 text-xs font-bold ${growth >= 0 ? "text-emerald-600" : "text-red-600"}`}>
   738→            {growth >= 0 ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   739→            {Math.abs(growth).toFixed(0)}%
   740→          </div>

   742→      </div>
   743→    </Card>
   744→  );
   745→  if (href) {
   746→    return <Link href={href}>{content}</Link>;
   747→  }
   748→  return content;
   749→}
